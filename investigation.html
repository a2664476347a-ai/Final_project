# Remove unnecessary columns and missing math scores
student_math_pro <- student_math |>
  select(-general_family_support) |>
  filter(
    if_all(
      starts_with("math_score_pv"),
      ~ !is.na(.)
    )
  )
# Select variables for data processing

student_processed <- student_math_pro |>
  select(
    student_id,
    country_code,
    student_age,
    student_gender,
    family_socioeconomic_status,
    family_support,
    home_possessions_index,
    home_ict_resources,
    highest_parental_education,
    parental_education_years,
    highest_parental_occupational_status,
    starts_with("math_score_pv")
  )
# calculate mean score for each student
student_processed <- student_processed |>
  mutate(
    math_score = rowMeans(
      across(starts_with("math_score_pv"))
    )
  ) |>
  select(-starts_with("math_score_pv"))
# Calculate statistics
score_stats <- student_processed %>%
  filter(!is.na(math_score)) %>%
  summarise(
    mean_score = mean(math_score),
    median_score = median(math_score),
    q1 = quantile(math_score, 0.25),
    q3 = quantile(math_score, 0.75)
  )
# Calculate the 2.5th and 97.5th percentiles
ci <- quantile(
  student_processed$math_score,
  probs = c(0.025, 0.975)

#correlation between conditions and math performance
correlation_data <- student_processed |>
    select(
        math_score,
        highest_parental_occupational_status,
        highest_parental_education,
        home_ict_resources
    )

correlation_values <- correlation_data |>
    summarise(
        across(
            -math_score,
            ~ cor(.x, math_score, use = "complete.obs")
        )
    ) |>
    pivot_longer(
        everything(),
        names_to = "Variable",
        values_to = "Correlation"
    )
ggplot(
    correlation_values,
    aes(
        x = "Mathematics Score",
        y = Variable,
        fill = Correlation
    )
) +
    geom_tile(
        colour = "black",
        linewidth = 0.8
    ) +
    geom_text(
        aes(label = round(Correlation, 2)),
        size = 4
    ) +
    scale_fill_gradient2(
        low = "#2878B5",
        mid = "white",

        midpoint = 0,

    labs(
        title = "Correlation with Mathematics Performance",
